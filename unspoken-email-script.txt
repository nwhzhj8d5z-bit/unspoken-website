var SITE_URL = "https://unspoken-website.vercel.app";
var WEB_APP_URL = "https://script.google.com/macros/s/AKfycbwSlNTSC2c1EM5B4yq7FhyDA5wM02H26aMuGpYX0CtcyJXUHe_sKF5Fya_u4Hvmg5NJrQ/exec";

function sheet_() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
}

function esc_(s) {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}

/* ---------- New subscriber: save + welcome email ---------- */
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var email = String(data.email || "").trim().toLowerCase();
  if (email) {
    var s = sheet_();
    var rows = s.getDataRange().getValues();
    var found = -1;
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim().toLowerCase() === email) { found = i; break; }
    }
    if (found === -1) {
      s.appendRow([email, new Date(), ""]);
      sendWelcome_(email);
    } else if (String(rows[found][2] || "").toLowerCase() === "unsubscribed") {
      s.getRange(found + 1, 3).setValue("");
      sendWelcome_(email);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

/* ---------- Unsubscribe link ---------- */
function doGet(e) {
  var email = e && e.parameter && e.parameter.unsubscribe;
  if (email) {
    var s = sheet_();
    var rows = s.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim().toLowerCase() === String(email).trim().toLowerCase()) {
        s.getRange(i + 1, 3).setValue("unsubscribed");
      }
    }
    return HtmlService.createHtmlOutput('<p style="font-family:Georgia,serif;padding:40px;font-size:18px;">You have been unsubscribed from Unspoken. Take care.</p>');
  }
  return HtmlService.createHtmlOutput("Unspoken subscribe endpoint.");
}

/* ---------- Email design ---------- */
function shell_(inner, email) {
  var unsub = WEB_APP_URL + "?unsubscribe=" + encodeURIComponent(email);
  return '<div style="max-width:560px;margin:0 auto;font-family:Georgia,serif;color:#111111;line-height:1.6;">'
    + '<div style="background:#111111;padding:14px 20px;margin-bottom:28px;"><span style="color:#FAFAFA;font-family:Arial,sans-serif;font-weight:bold;letter-spacing:3px;font-size:14px;">UNSPOKEN</span></div>'
    + inner
    + '<p style="font-family:Arial,sans-serif;font-size:12px;color:#777777;border-top:1px solid #E0E0E0;margin-top:36px;padding-top:14px;">You are receiving this because you subscribed to Unspoken. <a href="' + unsub + '" style="color:#777777;">Unsubscribe</a></p>'
    + '</div>';
}

function sendWelcome_(email) {
  var inner = '<p style="font-size:28px;font-style:italic;margin:0 0 12px;">You\'re in.</p>'
    + '<p>Thanks for subscribing to Unspoken, a magazine for the things our generation lives and no one names.</p>'
    + '<p>Whenever we publish something new, it will land right here in your inbox.</p>'
    + '<p><a href="' + SITE_URL + '" style="display:inline-block;background:#111111;color:#FAFAFA;padding:12px 22px;text-decoration:none;font-family:Arial,sans-serif;">Read the latest</a></p>'
    + '<p style="font-style:italic;">You\'ll relate to this.</p>';
  MailApp.sendEmail({ to: email, subject: "You're in. Welcome to Unspoken", htmlBody: shell_(inner, email), name: "Unspoken" });
}

function articleEmail_(a, email) {
  var link = SITE_URL + "/" + a.url;
  var img = a.image
    ? '<p><a href="' + link + '"><img src="' + SITE_URL + "/" + a.image + '" alt="" style="width:100%;max-width:560px;display:block;"></a></p>'
    : "";
  var inner = '<p style="font-family:Arial,sans-serif;font-size:12px;letter-spacing:2px;color:#4A6B4E;margin:0 0 6px;">NEW ARTICLE</p>'
    + '<p style="font-size:28px;font-style:italic;margin:0 0 6px;">' + esc_(a.title) + '</p>'
    + '<p style="font-family:Arial,sans-serif;font-size:13px;color:#777777;margin:0 0 18px;">By ' + esc_(a.author) + '</p>'
    + img
    + '<p>' + esc_(a.teaser) + '</p>'
    + '<p><a href="' + link + '" style="display:inline-block;background:#111111;color:#FAFAFA;padding:12px 22px;text-decoration:none;font-family:Arial,sans-serif;">Read it now</a></p>';
  return shell_(inner, email);
}

/* ---------- Automatic new-article emails ---------- */
function getLatestArticle_() {
  var res = UrlFetchApp.fetch(SITE_URL + "/articles.js?cb=" + new Date().getTime(), { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) return null;
  var text = res.getContentText().replace(/^\s*var\s+ARTICLES\s*=\s*/, "").replace(/;\s*$/, "");
  var list = JSON.parse(text);
  return list.length ? list[0] : null;
}

function activeSubscribers_() {
  var rows = sheet_().getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var email = String(rows[i][0]).trim();
    var status = String(rows[i][2] || "").toLowerCase();
    if (email && status !== "unsubscribed") out.push(email);
  }
  return out;
}

function sendArticle_(a, onlyMe) {
  var list = onlyMe ? [Session.getEffectiveUser().getEmail()] : activeSubscribers_();
  if (list.length > MailApp.getRemainingDailyQuota()) return -1;
  list.forEach(function (email) {
    MailApp.sendEmail({ to: email, subject: "New on Unspoken: " + a.title, htmlBody: articleEmail_(a, email), name: "Unspoken" });
  });
  return list.length;
}

function checkForNewArticle() {
  var latest = getLatestArticle_();
  if (!latest) return;
  var props = PropertiesService.getScriptProperties();
  var lastSent = props.getProperty("LAST_SENT_SLUG");
  if (lastSent === latest.slug) return;
  if (!lastSent) { props.setProperty("LAST_SENT_SLUG", latest.slug); return; }
  var result = sendArticle_(latest, false);
  var owner = Session.getEffectiveUser().getEmail();
  if (result >= 0) {
    props.setProperty("LAST_SENT_SLUG", latest.slug);
  } else if (props.getProperty("ALERTED_SLUG") !== latest.slug) {
    MailApp.sendEmail(owner, "Unspoken: new article emails not sent", "Your daily email limit is lower than your subscriber count, so nothing was sent. The script will keep trying every hour.");
    props.setProperty("ALERTED_SLUG", latest.slug);
  }
}

/* ---------- One-time setup and test ---------- */
function setup() {
  var latest = getLatestArticle_();
  if (latest) PropertiesService.getScriptProperties().setProperty("LAST_SENT_SLUG", latest.slug);
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger("checkForNewArticle").timeBased().everyHours(1).create();
  Logger.log("Done. Latest article recorded: " + (latest ? latest.title : "none found"));
}

function sendTestToMe() {
  var latest = getLatestArticle_();
  if (latest) sendArticle_(latest, true);
}
